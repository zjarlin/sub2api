```sql
SELECT e.id, r.request_id, r.requested_model, r.state, r.plan_id,
       c->>'reason' AS reason, c->>'eligible' AS eligible, count(*)
FROM ops_error_logs e
JOIN auto_model_routes r ON r.request_id = e.request_id::uuid
JOIN auto_model_route_plans p
  ON p.api_key_id = r.api_key_id AND p.session_id = r.session_id AND p.plan_id = r.plan_id
CROSS JOIN LATERAL jsonb_array_elements(p.candidates) c
WHERE e.id IN (198271,198272,198273,198274,198275)
GROUP BY e.id,r.request_id,r.requested_model,r.state,r.plan_id,c->>'reason',c->>'eligible'
ORDER BY e.id,count(*) DESC;
```
