SELECT e.id,r.state,r.requested_model,r.plan_id,c->>'reason' reason,count(*)
FROM ops_error_logs e
JOIN auto_model_routes r ON r.request_id::text=e.request_id
JOIN auto_model_route_plans p ON p.api_key_id=r.api_key_id
 AND p.session_id=r.session_id AND p.plan_id=r.plan_id
